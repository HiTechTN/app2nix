# ✅ CORRECTION — Utiliser QThread
class InstallWorker(QThread):
    finished = pyqtSignal(str)          # store_path
    failed   = pyqtSignal(str)          # error message
    progress = pyqtSignal(str)          # log lines

    def __init__(self, pkg_dir, nix_expr, env):
        super().__init__()
        self.pkg_dir = pkg_dir
        self.nix_expr = nix_expr
        self.env = env

    def run(self):
        proc = subprocess.Popen(
            ["nix-build", "default.nix"],
            cwd=str(self.pkg_dir),
            stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
            text=True, env=self.env
        )
        for line in proc.stdout:
            self.progress.emit(line.rstrip())
        proc.wait()
        if proc.returncode == 0:
            self.finished.emit(proc.stdout.read().strip())
        else:
            self.failed.emit(f"nix-build exited {proc.returncode}")
